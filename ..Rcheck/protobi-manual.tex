\HeaderA{protobi\_get\_table\_details}{Get data table details}{protobi.Rul.get.Rul.table.Rul.details}
%
\begin{Description}\relax
protobi\_get\_table\_details() returns details of all data tables associated with a Protobi project's schema (default) or primary-table.
\end{Description}
%
\begin{Usage}
\begin{verbatim}
protobi_get_table_details(
  projectid,
  apikey,
  scopekey = "schema",
  host = "https://v4.protobi.com",
  only_names = FALSE,
  prune = NULL
)
\end{verbatim}
\end{Usage}
%
\begin{Arguments}
\begin{ldescription}
\item[\code{projectid}] A character. Protobi project identifier.

\item[\code{apikey}] A character. The APIKEY from your account profile, https://app.protobi.com/account.

\item[\code{scopekey}] A character. Defaults to "schema", can also be "primary". Gathers either all schema-table names or all primary-table names.

\item[\code{host}] URL, defaults to "https://v4.protobi.com"

\item[\code{only\_names}] TRUE/FALSE, defaults to FALSE. Use TRUE to return only data table names. Optionally use 'prune' in tandem to filter result.

\item[\code{prune}] Optional stringr-formatted pattern. Pattern match data table names to remove from result. only\_names must be TRUE to use.
\end{ldescription}
\end{Arguments}
%
\begin{Value}
An R data frame
\end{Value}

%
\begin{Description}\relax
This function uploads data.frame content to Protobi
\end{Description}
%
\begin{Usage}
\begin{verbatim}
protobi_put_data(
  df,
  projectid,
  tablekey,
  apikey,
  host = "https://app.protobi.com",
  type = "data",
  filename = "data.csv",
  timeout_seconds = 300
)
\end{verbatim}
\end{Usage}
%
\begin{Arguments}
\begin{ldescription}
\item[\code{df}] A data.frame to uploaded to Protobi.

\item[\code{projectid}] A character. Protobi project identifier.

\item[\code{tablekey}] A character. The key of your Protobi data table.

\item[\code{apikey}] A character. The APIKEY from your account profile, https://app.protobi.com/account.

\item[\code{host}] URL, defaults to "https://app.protobi.com"

\item[\code{type}] Type of data, defaults to "data"

\item[\code{filename}] Optional name of file, defaults to "data.csv"

\item[\code{timeout\_seconds}] Maximum time in seconds to wait for async task completion, defaults to 300 (5 minutes)
\end{ldescription}
\end{Arguments}
%
\begin{Value}
A list with elements: complete (boolean), result (url when complete=true), and possibly callback (url) or message (string) when complete=false
\end{Value}
